Sandboxes ​

The sandbox packages: @chamvi/sandbox (contracts, rules, the workspace manager) and the three providers. See Workspaces & Coding for the behaviour.

@chamvi/sandbox ​

bash
npm install @chamvi/sandbox @chamvi/core ai --save

@chamvi/core and ai are peer dependencies.

createWorkspaceManager ​

ts
function createWorkspaceManager(options: WorkspaceManagerOptions): IWorkspaceManager;

const MAX_WORKSPACES_PER_RUN: 3;
const GC_INTERVAL_MS: 600_000;
const CLOSE_TIMEOUT_MS: 30_000;

Throws InvalidAgentDefinitionError for a provider without create and collectGarbage, an empty image, invalid env or limits, an allowRepo or onError that is no function, a maxWorkspacesPerRun below 1 or a negative gcIntervalMs.

ts
/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

import type { ToolSet } from 'ai';
import type { Engine, IClock } from '@chamvi/core';
import type { ISandboxProvider, SandboxLimits } from '../provider/types.ts';
import type { IWorkspace } from '../workspace/types.ts';

export type WorkspaceManagerOptions = {
    provider: ISandboxProvider;
    /** Passed to every `create`. */
    image?: string;
    /** Passed to every `create`. */
    env?: Record<string, string>;
    /** Passed to every `create`. */
    limits?: SandboxLimits;
    /** Accept `file:///absolute/path` repositories (local development and tests). Default false. */
    allowFileRepos?: boolean;
    /** Narrows the accepted repositories further; gets the normalized URL. */
    allowRepo?: (url: string) => boolean;
    /** Workspaces one run may hold at once. Default MAX_WORKSPACES_PER_RUN (3). */
    maxWorkspacesPerRun?: number;
    /** Minimum interval between the garbage collections `acquire` starts. Default GC_INTERVAL_MS; 0 disables them. */
    gcIntervalMs?: number;
    clock?: IClock;
    /** Background failures (disposal, garbage collection). Default console.error. */
    onError?: (error: unknown) => void;
};

export type AcquireInput = {
    /** `owner/name`, an https URL, or `file:///absolute/path` with `allowFileRepos`. */
    repo?: string;
    ref?: string;
    signal?: AbortSignal;
};

export type CloseOptions = {
    /** How long close() waits for disposals and creates in flight. Default CLOSE_TIMEOUT_MS (30 s). */
    timeoutMs?: number;
};

export type AcquireResult = {
    workspace: IWorkspace;
    /** True when the run already held (or was creating) the workspace of this repository. */
    reused: boolean;
};

/**
 * Run-scoped workspaces: one per run and repository (or none), created
 * through the provider and disposed when the run settles (`attach`).
 */
export interface IWorkspaceManager {
    /**
     * The run's workspace of `repo` (none: the repo-less one), created when
     * missing, expired or gone (`IWorkspace.disposed`). Concurrent calls share one create. ValidationError for a
     * repository or ref the rules refuse; SandboxError for another ref of a
     * held repository, a reached per-run cap or a failed create.
     */
    acquire(runId: string, input: AcquireInput): Promise<AcquireResult>;
    /**
     * The run's workspace by id, or its only live one when id is omitted
     * (expired and gone ones are disposed first). SandboxError: unknown,
     * expired or gone (disposed then), none; ValidationError: several and no id.
     */
    get(runId: string, workspaceId?: string): IWorkspace;
    /**
     * Disposes one workspace (SandboxError /not found/ for an id the run does
     * not hold, like get), or all of the run (never rejects for none); the
     * latter aborts the run's creates in flight and waits for them.
     */
    release(runId: string, workspaceId?: string): Promise<void>;
    /**
     * Disposes a run's workspaces on completed | failed | cancelled. Returns the
     * unsubscribe; after close() it subscribes nothing.
     */
    attach(engine: Pick<Engine, 'on'>): () => void;
    /** `create_workspace` and `dispose_workspace`. */
    tools(): ToolSet;
    /** Disposes the manager's expired workspaces, then runs the provider's collection; returns how many went. */
    collectGarbage(): Promise<number>;
    /**
     * Unsubscribes every attach, aborts the creates in flight and disposes
     * every workspace, waiting for both, for disposals started earlier and for
     * the creates in flight of runs released before (call after engine.drain()); afterwards
     * acquire rejects and attach subscribes nothing. Resolves after `timeoutMs`
     * at the latest, reporting what is still being disposed to `onError`.
     */
    close(options?: CloseOptions): Promise<void>;
}

export type WorkspaceToolResult = {
    workspace: string;
    root: '/workspace';
    repo: string | null;
    ref: string | null;
    commit: string | null;
    expiresAt: string;
    reused: boolean;
};

export type DisposeWorkspaceResult = {
    workspace: string;
    disposed: true;
};

ts
/*

/** What a provider knows about a workspace; `commit` is set after the checkout. */
export type WorkspaceInfo = {
    runId: string;
    /** `ISandboxProvider.name` of the provider that created it. */
    provider: string;
    /** The normalized repository URL. */
    repo?: string;
    ref?: string;
    /** HEAD after the checkout. */
    commit?: string;
    createdAt: Date;
    expiresAt: Date;
};

export type ExecOptions = {
    /** Relative to the workspace root or under `/workspace/`; default the root. */
    cwd?: string;
    /** Default `EXEC_TIMEOUT_MS`, at most `EXEC_TIMEOUT_MAX_MS`. */
    timeoutMs?: number;
    stdin?: string;
    signal?: AbortSignal;
};

export type ExecResult = {
    exitCode: number;
    /** At most `OUTPUT_MAX_BYTES` bytes of UTF-8. */
    stdout: string;
    stderr: string;
    stdoutTruncated: boolean;
    stderrTruncated: boolean;
    timedOut: boolean;
    durationMs: number;
};

/**
 * A disposable directory a run executes commands in. Exec takes argv only,
 * never a shell string.
 */
export interface IWorkspace {
    readonly id: string;
    readonly info: WorkspaceInfo;
    /**
     * True after `dispose()`, or once the workspace found its container, pod
     * or directory gone (stopped, killed or removed outside chamvi). Every
     * call then rejects with SandboxError; the manager drops such a workspace
     * and disposes it, which removes what is left of it.
     */
    readonly disposed: boolean;
    exec(argv: readonly string[], options?: ExecOptions): Promise<ExecResult>;
    /** UTF-8 text of at most `FILE_MAX_BYTES`. */
    readFile(path: string): Promise<string>;
    writeFile(path: string, content: string): Promise<void>;
    /** Idempotent. Every call after it rejects with SandboxError. */
    dispose(): Promise<void>;
}

normalizeRepo, assertGitRef and resolveWorkspacePath throw ValidationError (model input); resolveLimits, parseCpu, parseMemory and assertSandboxEnv throw InvalidAgentDefinitionError (configuration); prepareWorkspace throws SandboxError with git's stderr when a checkout step fails.

Remote workspaces ​

ts
function createRemoteWorkspace(input: RemoteWorkspaceInput): IWorkspace;

const TIMEOUT_KILL_AFTER_SECONDS: 5;
const EXEC_GRACE_MS: 15_000;
const ALIVE_PROBE_TIMEOUT_MS: 5_000;

A provider that can run argv in a container implements IExecTransport: it runs argv, streams stdout and stderr into the sinks it is given and resolves the exit code. createRemoteWorkspace adds the rest: the sh -c script that enters the working directory, runs the command behind the timeout wrapper and reports its exit on stderr (so an exec returns about 2 s after the command exited, even when a background job holds its output), the guard, path checks, output caps, file reads and writes, dispose. After a failed exec it asks alive() (at most ALIVE_PROBE_TIMEOUT_MS); false marks the workspace gone (disposed).

ts
/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

import type { IClock } from '@chamvi/core';
import type { IOutputSink } from '../output/types.ts';
import type { WorkspaceInfo } from '../workspace/types.ts';

export type TransportExecOptions = {
    stdin?: Uint8Array;
    /** Gets every stdout chunk as it arrives; the workspace caps and keeps it. */
    stdout: IOutputSink;
    /** Gets every stderr chunk as it arrives; the workspace caps and keeps it. */
    stderr: IOutputSink;
    /**
     * Aborted on a caller abort, a dispose, the client-side guard, or once the
     * workspace settled the exec itself (the command exited, a background job
     * still holds its output): stop waiting and detach.
     */
    signal: AbortSignal;
};

export type TransportExecResult = {
    exitCode: number;
};

/**
 * Runs argv in one container or pod. The remote workspace wraps argv itself
 * (the working directory, GNU `timeout`, the exit status reported on
 * stderr); a transport only executes and streams the output.
 */
export interface IExecTransport {
    /** A command that cannot start reports exit 127. */
    exec(argv: readonly string[], options: TransportExecOptions): Promise<TransportExecResult>;
    /**
     * Whether the container or pod still runs; asked after a failed exec. False
     * marks the workspace gone. A rejection, or no answer within
     * `ALIVE_PROBE_TIMEOUT_MS` (the signal then aborts), reads as alive.
     */
    alive?(signal: AbortSignal): Promise<boolean>;
    /** Deletes the container or pod; idempotent, a missing one is fine. */
    remove(): Promise<void>;
}

export type RemoteWorkspaceInput = {
    /** `randomUUID()`. */
    id: string;
    /** A mutable object the provider owns (it sets `commit` after the checkout). */
    info: WorkspaceInfo;
    transport: IExecTransport;
    /** Default `systemClock`. */
    clock?: IClock;
};

Context ​

function provideSandboxContext(provider: IContextProvider, manager: IWorkspaceManager): void; // engine.provide under the key 'sandbox'

sandboxModule() provides its manager; with a hand-built engine call provideSandboxContext(engine, manager) after new Engine(). A tools factory then builds the workspace and coding tools per execution: tools: (ctx) => { const manager = injectSandboxContext(ctx); return { ...manager.tools(), ...codingTools(manager) }; }. See Context.

Module ​

ts
// @chamvi/sandbox/module, needs @chamvi/app
function sandboxModule(options?: SandboxModuleOptions): SandboxModule; // name SANDBOX_MODULE_NAME 'chamvi:sandbox', .manager: IWorkspaceManager

const SANDBOX_PROVIDER_NAMES: readonly SandboxProviderName[]; // ['local', 'docker', 'kubernetes']

The provider is CHAMVI_SANDBOX_PROVIDER, else options.provider, else local, read from options.env ?? process.env when the factory is called; CHAMVI_SANDBOX_NAMESPACE and CHAMVI_SANDBOX_IMAGE win over the options too. In start and job mode setup imports the provider package (@chamvi/sandbox-local, @chamvi/sandbox-docker or @chamvi/sandbox-k8s, installed beside @chamvi/sandbox and resolved from there; they are no declared peers, since every provider depends on @chamvi/sandbox and release-please refuses the cycle; a missing one throws InvalidAgentDefinitionError naming it, and under a strict layout that hides undeclared packages (Yarn PnP, pnpm with hoist=false) pass a built provider as options.provider), and the lifecycle hooks attach the manager to the engine, provide it (engineCreated) and close it after the engine drained; migrate mode does nothing. @chamvi/app is an optional peer: the main entry never imports it.

ts
export type SandboxProviderName = 'local' | 'docker' | 'kubernetes';

/**
 * `createWorkspaceManager()` options; the manager's `env` is `workspaceEnv`
 * here, since `env` is the process env the factory reads.
 */
export type SandboxModuleOptions = Omit<WorkspaceManagerOptions, 'provider' | 'env'> & {
    /** A provider name (`CHAMVI_SANDBOX_PROVIDER` wins) or an instance; default `'local'`. */
    provider?: SandboxProviderName | ISandboxProvider;
    /** Used as is; no provider is built then. */
    manager?: IWorkspaceManager;
    /** Kubernetes namespace of the workspace pods; `CHAMVI_SANDBOX_NAMESPACE` wins. */
    namespace?: string;
    /** Docker and Kubernetes image; `CHAMVI_SANDBOX_IMAGE` wins, default `DEFAULT_SANDBOX_IMAGE`. */
    image?: string;
    /** Added to every workspace's environment. */
    workspaceEnv?: Record<string, string>;
    /** Default `process.env`, read when the factory is called. */
    env?: Env;
};
